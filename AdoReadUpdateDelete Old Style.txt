public class HomeController : Controller
{
    SqlConnection db;
    SqlCommand cm;
    SqlDataReader dr;
    string st = "";
    string con = "";
    public ActionResult Index()
    {
        return View();
    }

    public ActionResult Show(IFormCollection fc)
    {
        con = @"Server=DESKTOP-7145BT3\SQLEXPRESS;Database=Lcc;Integrated Security=True;TrustServerCertificate=True";
        db = new SqlConnection();
        if (db.State == ConnectionState.Open)
        {
            db.Close();
        }
        db.ConnectionString = con;
        db.Open();
        st = "select eno,ename,doj,sal from staff where eno=" + fc["En"] + "";
        cm = new SqlCommand(st, db);
        dr = cm.ExecuteReader();
        if (dr.Read() == true)
        {
            Employee obj = new Employee();
            obj.Eno = Convert.ToInt32(dr["eno"]);
            obj.Ename = dr["ename"].ToString();
            obj.Doj = Convert.ToDateTime(dr["Doj"]);
            obj.Sal = Convert.ToInt32(dr["sal"]);
            ViewBag.msg = "Record Shown";
            return View(obj);
        }
        else
        {
            ViewBag.msg = "Eno Not Found";
            return View("Index");
        }

    }

    public ActionResult MyAction(string submitButton, IFormCollection f)
 {

     switch (submitButton)
     {
         case "Modify":
            Modify(f);
            ViewBag.Message = "Record Modified";
            return View();
        
             case "Delete":
             Delete(f);
              ViewBag.Message = "Record Modified";
             return  View();

     }
    return View();
 }



    public ActionResult Modify(IFormCollection fc)
    {
        con = @"Server=DESKTOP-7145BT3\SQLEXPRESS;Database=Lcc;Integrated Security=True;TrustServerCertificate=True";
        db = new SqlConnection();
        if (db.State == ConnectionState.Open)
        {
            db.Close();
        }
        db.ConnectionString = con;
        db.Open();
        st = "update staff set ename='" + fc["Ename"] + "',Doj='" + fc["Doj"] + "',Sal=" + fc["Sal"] + " where eno=" + fc["Eno"] + "";
        cm = new SqlCommand(st, db);
        cm.ExecuteNonQuery();
        return View("Modify");
    }


    public ActionResult Delete(IFormCollection fc)
    {
        con = @"Server=DESKTOP-7145BT3\SQLEXPRESS;Database=Lcc;Integrated Security=True;TrustServerCertificate=True";
        db = new SqlConnection();
        if (db.State == ConnectionState.Open)
        {
            db.Close();
        }
        db.ConnectionString = con;
        db.Open();
        st = "Delete from Staff where eno=" + fc["Eno"] + "";
        cm = new SqlCommand(st, db);
        cm.ExecuteNonQuery();
        return View("Delete");
    }
}
